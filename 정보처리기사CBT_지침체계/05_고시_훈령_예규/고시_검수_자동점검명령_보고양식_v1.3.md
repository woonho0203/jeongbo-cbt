## 제1조 JSON 파싱·필수 필드 점검

> `answer`는 정수뿐 아니라 전항정답·복수정답 배열도 정상이다(데이터관리법 제2조 6호). 배열이면 모든 원소가 범위 안인지 검사한다. 배열을 무조건 오류로 잡던 v1.1 로직은 정상 복수정답 8건을 오탐했다(결정례_2026헌가3).

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const dataDir = 'data';
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
let errors = [];
for (const file of fs.readdirSync(dataDir).filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const full = path.join(dataDir, file);
  let json;
  try { json = JSON.parse(fs.readFileSync(full, 'utf8')); }
  catch (error) { errors.push(`${file}: JSON 파싱 실패 - ${error.message}`); continue; }
  const questions = Array.isArray(json) ? json : json.questions;
  if (!Array.isArray(questions)) { errors.push(`${file}: questions 배열 없음`); continue; }
  for (const q of questions) {
    const label = `${file} Q${q.qnum ?? '?'}`;
    if (!q.stem) errors.push(`${label}: stem 없음`);
    if (!Array.isArray(q.options) || q.options.length < 2) errors.push(`${label}: options 이상`);
    const ans = Array.isArray(q.answer) ? q.answer : [q.answer];
    const answerBad = ans.length === 0 ||
      ans.some(a => !Number.isInteger(Number(a)) || Number(a) < 1 || Number(a) > q.options.length);
    if (answerBad) errors.push(`${label}: answer 범위 이상 (값: ${JSON.stringify(q.answer)})`);
    if (!String(q.explanation || '').trim()) errors.push(`${label}: explanation 없음`);
  }
}
if (errors.length) { console.log(errors.join('\n')); process.exit(1); }
console.log('필수 필드 점검 통과');
NODE
```

## 제2조 짧거나 불충분한 해설 후보 찾기

> 부정형 판단 토큰은 "아니다/않는다/없다"까지 인정한다. 좁은 토큰 목록(`틀린|아닌|옳지|잘못|거리가 먼`)은 '~이 아니다'로 끝나는 정상 부정형 해설을 오탐했다(결정례_2026헌가4).

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
const weak = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  for (const q of questions) {
    const exp = String(q.explanation || '').replace(/\s+/g, ' ').trim();
    const negative = /(틀린 것은|맞지 않은 것은|아닌 것은|옳지 않은 것은|해당하지 않는|볼 수 없는|없는 것은|거리가 먼 것은|적절하지 않은)/.test(q.stem || '');
    const hasSpecificValue = /[0-9]|→|=|출력|결과|공식|행|열|순서|값|개월|번/.test(exp);
    if (exp.length < 45 || (negative && !/(틀린|아닌|아니|않|없|옳지|잘못|거리가 먼)/.test(exp)) || !hasSpecificValue)
      weak.push(`${file} Q${q.qnum}: ${exp.slice(0, 120)}`);
  }
}
console.log(weak.length ? weak.join('\n') : '불충분 후보 없음');
NODE
```

## 제2조의2 무맥락 템플릿 후보 찾기 (해설작성법 제5조 1호·판례_001)

> 길이·값만 보는 제2조는 템플릿 후렴구가 든 긴 해설을 놓친다. 아래는 도메인 지식 없이 어느 문제에나 붙는 금지 문구(고시_해설형식 제4조)의 실제 잔존을 직접 탐지한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
const templates = [
  /문제에서 찾는 예외가 아니라 해당 범주에 속하거나/,
  /정답 조건에서 제외됨/,
  /문제에서 묻는 조건과 맞지 않으므로 틀림/,
  /최종 출력값 또는 계산 결과에 해당하므로 정답임/,
  /문제 단서와 다른 개념이므로 정답이 아님/,
  /계산하거나 표와 대조했을 때 도출되는 값임/,
  /문제 단서와 비교해야 하는 핵심 선택지 개념임/,
];
let total = 0; const perFile = {};
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  let c = 0;
  for (const q of questions) {
    const e = String(q.explanation || '');
    if (templates.some(p => p.test(e))) { c++; total++; }
  }
  if (c) perFile[file] = c;
}
for (const [f, c] of Object.entries(perFile)) console.log(`${f}: ${c}`);
console.log(`무맥락 템플릿 총계: ${total}`);
NODE
```

## 제3조 이미지·표 해설 연계 점검

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
const suspects = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  for (const q of questions) {
    if (!q.image && !q.table) continue;
    const exp = String(q.explanation || '');
    if (!/(그림|표|행|열|노드|간선|자식|순회|방문|기준 주소|길이|변위|출력|결과|값|=|[0-9])/.test(exp))
      suspects.push(`${file} Q${q.qnum}: 이미지/표 값 언급 부족`);
  }
}
console.log(suspects.length ? suspects.join('\n') : '이미지·표 해설 연계 후보 없음');
NODE
```

## 제4조 이미지·표 삽입 현황·경로·HTML 점검

> 점검 범위는 검수법 제2조의 검수 대상 전체(회차별 기출 `exam_*` + 유형별 `category_*`)다.
> 특정 연도만 점검하면 안 된다. (2026헌가2: 2016~2023 한정 점검으로 2024·2025와 유형별 이미지가 무검사 통과된 결함 시정)

이미지·표 개수 요약:

```bash
node - <<'NODE'
const fs = require('fs');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync('data/' + file, 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  const img = questions.filter(q => q.image).map(q => q.qnum);
  const table = questions.filter(q => q.table).map(q => q.qnum);
  console.log(`${file}: image ${img.length} [${img.join(', ')}] / table ${table.length} [${table.join(', ')}]`);
}
NODE
```

이미지 파일 누락 확인:

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
let missing = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f))) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  for (const q of questions) {
    if (q.image && !fs.existsSync(path.join('public', q.image))) missing.push(`${file} Q${q.qnum}: ${q.image}`);
  }
}
if (missing.length) { console.log(missing.join('\n')); process.exit(1); }
console.log('IMAGE PATH OK');
NODE
```

표 HTML 형식 확인:

```bash
node - <<'NODE'
const fs = require('fs');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
let bad = [];

NODE
```
