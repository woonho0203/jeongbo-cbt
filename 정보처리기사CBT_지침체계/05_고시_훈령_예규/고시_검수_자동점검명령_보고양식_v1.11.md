## 제1조 JSON 파싱·스키마·인덱스 점검

> `answer`는 정수뿐 아니라 전항정답·복수정답 배열도 정상이다(데이터관리법 제2조 8호). 배열이면 모든 원소가 범위 안이고 서로 다른지 검사한다. 회차 문항의 `subjectName`은 앱 추론에 맡기지 않고 JSON에 명시하며, 두 인덱스의 식별자·제목·문항 수도 실제 파일과 대조한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const dataDir = 'data';
const sourceFiles = fs.readdirSync(dataDir).filter(f => /^(exam|category)_.*\.json$/.test(f)).sort();
const subjectNames = {
  1: '소프트웨어 설계',
  2: '소프트웨어 개발',
  3: '데이터베이스 구축',
  4: '프로그래밍 언어 활용',
  5: '정보시스템 구축 관리',
};
const errors = [];
const actual = { exam: new Map(), category: new Map() };

for (const file of sourceFiles) {
  const full = path.join(dataDir, file);
  let json;
  try { json = JSON.parse(fs.readFileSync(full, 'utf8')); }
  catch (error) { errors.push(`${file}: JSON 파싱 실패 - ${error.message}`); continue; }
  const kind = file.startsWith('exam_') ? 'exam' : 'category';
  const idKey = kind === 'exam' ? 'examId' : 'categoryId';
  const expectedId = file.replace(kind + '_', '').replace(/\.json$/, '');
  if (json[idKey] !== expectedId) errors.push(`${file}: ${idKey} 불일치 (${json[idKey]} != ${expectedId})`);
  if (!String(json.title || '').trim()) errors.push(`${file}: title 없음`);

  const questions = json.questions;
  if (!Array.isArray(questions)) { errors.push(`${file}: questions 배열 없음`); continue; }
  actual[kind].set(json[idKey], { title: json.title, count: questions.length, file });
  const seenQnums = new Set();

  for (const q of questions) {
    const label = `${file} Q${q.qnum ?? '?'}`;
    if (!Number.isInteger(q.qnum) || q.qnum < 1) errors.push(`${label}: qnum 이상`);
    if (seenQnums.has(q.qnum)) errors.push(`${label}: qnum 중복`);
    seenQnums.add(q.qnum);
    if (!String(q.stem || '').trim()) errors.push(`${label}: stem 없음`);
    if (!Array.isArray(q.options) || q.options.length < 2 || q.options.some(o => typeof o !== 'string')) {
      errors.push(`${label}: options 이상`);
    }
    const answerArray = Array.isArray(q.answer);
    const ans = answerArray ? q.answer : [q.answer];
    const answerBad = ans.length === 0 ||
      ans.some(a => !Number.isInteger(a) || a < 1 || a > (q.options || []).length) ||
      new Set(ans).size !== ans.length;
    if (answerBad || (!answerArray && !Number.isInteger(q.answer))) {
      errors.push(`${label}: answer 형식·범위 이상 (값: ${JSON.stringify(q.answer)})`);
    }
    if (!String(q.explanation || '').trim()) errors.push(`${label}: explanation 없음`);

    if (kind === 'exam') {
      if (!Number.isInteger(q.subject) || !subjectNames[q.subject]) errors.push(`${label}: subject 이상`);
      if (q.subjectName !== subjectNames[q.subject]) {
        errors.push(`${label}: subjectName 불일치·누락 (${JSON.stringify(q.subjectName)} != ${JSON.stringify(subjectNames[q.subject])})`);
      }
    } else if (q.subject != null || q.subjectName != null) {
      if (!Number.isInteger(q.subject) || q.subjectName !== subjectNames[q.subject]) {
        errors.push(`${label}: 선택 과목 메타 불일치`);
      }
    }
  }
}

for (const [kind, indexFile, idKey] of [
  ['exam', 'exams_index.json', 'examId'],
  ['category', 'categories_index.json', 'categoryId'],
]) {
  let index;
  try { index = JSON.parse(fs.readFileSync(path.join(dataDir, indexFile), 'utf8')); }
  catch (error) { errors.push(`${indexFile}: JSON 파싱 실패 - ${error.message}`); continue; }
  if (!Array.isArray(index)) { errors.push(`${indexFile}: 배열 아님`); continue; }
  const seenIds = new Set();
  for (const item of index) {
    const id = item[idKey];
    if (seenIds.has(id)) errors.push(`${indexFile}: ${idKey} 중복 ${id}`);
    seenIds.add(id);
    const source = actual[kind].get(id);
    if (!source) { errors.push(`${indexFile}: 대응 파일 없음 ${id}`); continue; }
    if (item.title !== source.title) errors.push(`${indexFile}: ${id} title 불일치`);
    if (item.count !== source.count) errors.push(`${indexFile}: ${id} count ${item.count} != ${source.count}`);
  }
  for (const id of actual[kind].keys()) {
    if (!seenIds.has(id)) errors.push(`${indexFile}: 항목 누락 ${id}`);
  }
}

if (errors.length) { console.log(errors.join('\n')); process.exit(1); }
console.log(`스키마·인덱스 점검 통과 (${sourceFiles.length}개 파일)`);
NODE
```

## 제1조의2 보기 상호 구별 가능성 점검

> 제1조는 `options`가 **존재하고 2개 이상**인지만 본다. 그래서 보기 값이 공백이거나 서로 완전히 같아도 통과한다. 실제로 수식·도형 보기가 PDF에서 텍스트로 추출되지 않아 빈 문자열이나 중복 문자열로 저장된 문항이 이 사각지대로 무검사 통과했다 (판례_007). 보기는 서로 구별될 수 있어야 문항이 성립한다.
>
> **공백만 제거한 비교로는 부족하다.** exam_2019-1 Q26은 오버바가 소실돼 보기 ①·②가 시각적으로 동일해졌는데, 문자열은 `T = AㆍB +AㆍB`와 `T = AㆍB + AㆍB`로 **공백 한 칸만 달라** 초기 점검을 빠져나갔다. 따라서 공백·기호를 모두 제거한 **정규화 키**로도 함께 비교한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
const trim = s => String(s).replace(/\s+/g, ' ').trim();
// 공백만 걷어낸 정규화 키. 괄호·연산자는 수식에서 의미를 가지므로 제거하면 안 된다
// (`2^(n-1)`과 `2^n-1`이 같아져 정상 보기를 오탐한다 — 2026헌가9 실측).
const key = s => String(s).replace(/\s+/g, '');
const bad = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  for (const q of (Array.isArray(json) ? json : json.questions || [])) {
    const opts = (q.options || []).map(trim);
    const empty = opts.filter(o => o.length === 0).length;
    const dupRaw = opts.length - new Set(opts).size;
    const dupKey = opts.length - new Set(opts.map(key)).size;
    if (empty || dupRaw || dupKey) {
      const tag = dupKey > dupRaw ? ' [정규화 후에만 중복 — 기호 소실 의심]' : '';
      bad.push(`${file} Q${q.qnum}: 공백 ${empty} / 중복 ${dupRaw} / 정규화중복 ${dupKey}${tag}${q.image ? ' (image 보완됨)' : ''}`);
    }
  }
}
if (bad.length) { console.log(bad.join('\n')); console.log(`\n합계 ${bad.length}건`); process.exit(1); }
console.log('보기 상호 구별 점검 통과');
NODE
```

적발된 문항은 PDF 원본을 고배율(3~12배)로 렌더해 **눈으로 판독**한다. 텍스트 레이어가 비어 있어도 대부분 선형 텍스트로 전사할 수 있으므로 `options`를 복원하고, 그림·도표처럼 텍스트로 옮길 수 없는 자료만 `image` 필드로 보완한다 (추출·삽입법 제3조).

**정규화중복 판정 시 예외**: 공백 자체가 정답을 가르는 문항은 오탐이다. 파이썬 `print(a, b)`의 출력을 묻는 문제에서 보기 `53 44`와 `53t44`처럼 **공백 유무가 정답 근거**인 경우가 이에 해당한다(category_code Q30·Q34, exam_2022-3 Q65, exam_2025-1 Q65, exam_2025-3 Q73 — 2026-07-19 확인). 파이썬 `for`문의 원소별 줄바꿈 출력을 묻는 문제에서 보기 `대한민국`과 `대\n한\n민\n국`처럼 **줄바꿈 유무가 정답 근거**인 경우도 같다(exam_2022-1 Q79 — 2026-07-25 확인). 이런 문항은 `dupRaw`가 0이면 정상이므로 보고서에 예외로 기록하고 넘어간다.

## 제1조의3 독립 원본 대응 점검 (JSON ↔ 자기 PDF)

> 제1~4조는 모두 **파일 내부의 구조적 정합성**만 본다. 파일이 자기 독립 원본 PDF와 실제로 대응하는지는 별도 대조가 필요하다. `exam_2022-1.json`이 `exam_2022-2.json` 내용을 담고도 구조 점검을 통과한 사고가 있었고, 유형별 데이터도 회차 데이터의 사본이 아니라 각 유형별 PDF를 독립 원본으로 삼는다는 사용자 유권해석이 확정됐다. 이 조는 대량 데이터 작업 직후 실행하는 권장 절차이며 Python(`pdfplumber`)과 원본 PDF가 필요하다.

절차:
1. PDF 매핑 고시에서 각 JSON의 자기 PDF를 찾는다. `exam_*.json`은 회차별 PDF, `category_*.json`은 유형별 PDF를 사용한다.
2. `stem` 앞부분(공백 제거, 10~20자)을 자기 PDF 전체 텍스트 안에서 찾아 자기 일치율을 계산한다.
3. 자기 일치율이 낮으면 같은 방식으로 다른 PDF와 교차 대조해 오염 가능성을 찾되, 다른 파일과 비슷하다는 사실만으로 복사·교체하지 않는다.
4. 회차 파일이 다른 회차 PDF와 현저히 더 잘 맞으면 examId 오염 후보로 기록하고 자기 회차 PDF에서 재추출한다.
5. 유형별 파일은 비슷한 회차 문항이 있어도 자기 유형별 PDF를 기준으로 원문·보기·정답·자료를 확정한다. 회차 JSON은 검색 단서일 뿐 원본이 아니다.
6. 자기 일치율이 낮아도 다른 PDF와 유의미하게 더 맞지 않으면 이미지 보기·특수 글리프 등 구조적 원인을 확인하고 판정을 보류한다.

**알려진 오탐 원인** (exam_2016-3·2018-1 30여 문항 육안 재검토로 확인, 판례_실패_회차데이터오염_008): 아래에 해당하면 자기 일치율이 낮게 나와도 오염이 아니라 비교 스크립트의 한계다. 매번 육안 재검토를 반복하지 말고 이 목록으로 먼저 걸러낸다.
- 보기에 전각 기호(＞＝·＜＝·＋)나 화살표(⟵) 등 PDF 원본 폰트의 특수 글리프가 쓰인 경우 — 추출 텍스트와 문자 코드가 미세하게 달라 문자열 비교가 실패한다.
- 스템에 곱은따옴표(" ")가 쓰인 경우 — 정규화 과정에서 따옴표를 제거하지 않으므로 PDF의 직선 따옴표(")와 불일치한다.
- 보기가 다이어그램·도형인 2016년대 초기 PDF — 보기 자체가 텍스트가 아니라 이미지라 원천적으로 비교 대상이 없다(제1조의2 문제가 아니라 원본 PDF 구조 문제).

## 제1조의4 유형별 연습 독립 원본 점검

> 유형별 연습은 회차 데이터와 비슷한 문항을 포함할 수 있지만, 원문·보기·정답·이미지·표를 확정하는 근거는 대응하는 유형별 PDF다. 과거의 회차↔유형 동기화 검사는 후보 탐색 도구로는 쓸 수 있어도, 회차 데이터를 유형별 파일에 복사하는 자동 수정 근거로는 사용할 수 없다.

절차:

1. PDF 매핑 고시의 유형별 5개 매핑을 기준으로 `categoryId`·제목·문항 수·번호 연속성을 확인한다.
2. 각 문항의 `stem`·`options`·`answer`를 자기 유형별 PDF와 대조한다.
3. `image`·`table`이 있는 문항은 같은 유형별 PDF의 해당 자료와 대조한다.
4. 회차 데이터와 유사한 문항을 찾았더라도 차이가 있으면 유형별 PDF 표기를 우선한다.
5. 기존 회차 이미지 파일을 공유하는 문항은 두 PDF 자료가 동일한지 확인하고, 다르거나 확인할 수 없으면 `확인 필요`로 기록한다.
6. 판례_010에서 회차 데이터로 복원·교체했던 문항은 Git 이력으로 재확정한 17문항이며, 2026-07-26 자기 유형별 PDF 재대조를 완료했다. 이후 이 문항을 수정할 때도 회차 데이터가 아니라 자기 유형별 PDF를 직접 대조한다.

## 제2조 짧거나 불충분한 해설 후보 찾기

> 부정형 판단 토큰은 "아니다/않는다/없다"까지 인정한다. 좁은 토큰 목록(`틀린|아닌|옳지|잘못|거리가 먼`)은 '~이 아니다'로 끝나는 정상 부정형 해설을 오탐했다(결정례_2026헌가4).
> 활용형(`틀리며|틀리다|틀렸으며|거리가 멀다|거리가 멀며`)도 반드시 포함한다. 어간만 인정하던 v1.3 로직은 "정답 3번은 ~ 틀렸으며"처럼 정상적으로 틀린 이유를 서술한 부정형 해설 다수를 불충분 후보로 오탐했다(2026헌가6, 검수보고_2026-07-19에서 242건 중 다수 확인).
> 관형형 `틀린`도 별도 토큰으로 포함한다. "~2번이 틀린 설명이라 정답은 2번이다"처럼 매우 흔한 서술에 쓰이는데, `틀리`는 리≠린이라 문자열로 매칭되지 않아 v1.6까지 정상 해설 다수를 불충분 후보로 오탐했다(exam_2022-1 재추출 100건 검수 중 9건 실측, 2026헌가11).

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);
const weak = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  for (const q of questions) {
    const exp = String(q.explanation || '').replace(/\s+/g, ' ').trim();
    const negative = /(틀린 것은|맞지 않은 것은|아닌 것은|옳지 않은 것은|해당하지 않는|볼 수 없는|없는 것은|거리가 먼 것은|적절하지 않은)/.test(q.stem || '');
    const hasSpecificValue = /[0-9]|→|=|출력|결과|공식|행|열|순서|값|개월|번/.test(exp);
    const negTokenHit = /(틀리|틀린|틀렸|틀림|아니|않|없|옳지|잘못|거리가\s?먼|거리가\s?멀)/.test(exp);
    if (exp.length < 45 || (negative && !negTokenHit) || !hasSpecificValue)
      weak.push(`${file} Q${q.qnum}: ${exp.slice(0, 120)}`);
  }
}
console.log(weak.length ? weak.join('\n') : '불충분 후보 없음');
NODE
```

## 제2조의2 무맥락 템플릿 후보 찾기 (해설작성법 제5조 1호·판례_001)

> 길이·값만 보는 제2조는 템플릿 후렴구가 든 긴 해설을 놓친다. 아래는 도메인 지식 없이 어느 문제에나 붙는 금지 문구(고시_해설형식 제4조)의 실제 잔존을 직접 탐지한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
```

## 제2조의4 과목 교차 재사용 정의 탐지 (해설 교차 문항 오염)

> 제2조의2·제2조의3은 **내용이 공허한** 무맥락 템플릿만 잡는다. 이번에 발견된 유형은 반대로 **내용은 있지만 다른 문항 것**이다 — 예를 들어 HDLC 프레임 종류를 묻는 문항의 해설에 "응집도는 모듈 내부 요소의 관련 정도이고 결합도는…"이라는, 전혀 다른 주제(SW 설계)의 개념 정의가 그대로 붙어 있었다(헌법 제5조 2호 위반). 완전히 동일한 "개념 정의"·오답 분석 문구가 **공식 과목 분류(subject 1~5) 3개 이상**에 걸쳐 재사용되면 교차 오염 후보다 — 같은 주제라면 보통 과목 1~2개 안에 머무른다(판례_실패_해설교차문항재사용오염_009).

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');

      if (clause.length < 12) continue;
      if (!clauseMap.has(clause)) clauseMap.set(clause, new Set());
      clauseMap.get(clause).add(subj);
    }
  }
}
```

## 제4조의2 렌더링 출력 점검 (보기 셔플 재매핑)

> 제1~4조는 모두 **원본 JSON**을 검사한다. 그러나 앱은 매 세션 보기를 무작위로 섞고(`public/js/exam.js`) 해설 속 번호를 `remapExplanationNumbers()`로 바꿔서 화면에 그린다. 데이터가 정확해도 이 표시 단계에서 왜곡될 수 있으므로, **재매핑을 통과한 최종 출력**을 따로 검사해야 한다. 이 사각지대 때문에 376문항이 화면에 중복·소실된 번호를 표시하는 결함이 장기간 무검사 통과했다 (판례_006).
>
> 검사 원리: 해설이 참조하는 보기 번호 집합을 뽑아, 24개 순열 전부에 대해 `remap` 결과가 기대 집합과 일치하는지 확인한다. 앱의 `remapExplanationNumbers`가 개정되면 아래 사본도 함께 갱신한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);

// public/js/util.js의 remapExplanationNumbers 사본 (개정 시 동기화할 것)
function remap(text, sm) {
  if (!sm || !sm.length) return text;
  const inv = new Array(sm.length);
  sm.forEach((o, p) => { inv[o] = p + 1; });
  const c = ['①','②','③','④','⑤'];
  const expanded = text.replace(/[1-4](?:\s*[·,、]\s*[1-4])+\s*번(?!째)/g, (m) => {
    const nums = m.match(/[1-4]/g);
    const sep = m.includes('·') ? '·' : m.includes('、') ? '、' : ', ';
    return nums.map(n => n + '번').join(sep);
  });
  return expanded.replace(/[①②③④⑤]|([1-4])번(?!째)/g, (m, n) => {
    const o = (n !== undefined) ? parseInt(n) : (c.indexOf(m) + 1);
    const nn = inv[o - 1];
    if (nn == null) return m;
    return (n !== undefined) ? nn + '번' : (c[nn - 1] || m);
  });
}
function refNums(t) {
  const e = t.replace(/[1-4](?:\s*[·,、]\s*[1-4])+\s*번(?!째)/g,
    (m) => m.match(/[1-4]/g).map(n => n + '번').join('·'));
  const out = []; const re = /([1-4])번(?!째)/g; let m;
  while ((m = re.exec(e))) out.push(parseInt(m[1]));
  return out.sort().join(',');
}
function perms(a) {
  if (a.length <= 1) return [a];
  const r = [];
  a.forEach((v, i) => perms([...a.slice(0, i), ...a.slice(i + 1)]).forEach(p => r.push([v, ...p])));
  return r;
}
const ALL = perms([0, 1, 2, 3]);

let checked = 0; const bad = [];
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const j = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  for (const q of (Array.isArray(j) ? j : j.questions || [])) {
    const e = String(q.explanation || '');
    if (!/[1-4]번/.test(e) || !q.options || q.options.length !== 4) continue;
    const before = refNums(e);
    if (!before) continue;
    checked++;
    for (const sm of ALL) {
      const inv = new Array(4); sm.forEach((o, p) => { inv[o] = p + 1; });
      const expect = before.split(',').map(n => inv[parseInt(n) - 1]).sort((a, b) => a - b).join(',');
      if (refNums(remap(e, sm)) !== expect) { bad.push(`${file} Q${q.qnum} (셔플 [${sm}])`); break; }
    }
  }
}
console.log(`보기번호 참조 문항: ${checked}`);
if (bad.length) { console.log('재매핑 오작동:\n' + bad.join('\n')); process.exit(1); }
console.log('렌더링 출력 점검 통과 (24개 순열 전수)');
NODE
```

## 제4조의3 지문(stem)-표(table) 데이터 중복 탐지

> 제4조는 `table` 필드의 형식(HTML·이미지 경로)만 본다. `table` 필드를 나중에 추가·복사해 넣어도 `stem`에 원본 표 텍스트가 그대로 남아 있으면, 화면에 같은 자료가 **뭉쳐진 텍스트로 한 번, 정상 표로 또 한 번** 중복 표시된다(판례_실패_지문표데이터중복_011). `table`이 있고 `stem`에 개행이 있는 문항만 대상으로, 표 셀 텍스트(공백 제거)가 개행 이후 부분에 그대로 포함되는지 검사한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);

function tableCells(html) {
  const cells = [...html.matchAll(/<t[dh][^>]*>([\s\S]*?)<\/t[dh]>/g)].map(m => m[1].replace(/<[^>]+>/g, '').trim());
  return cells.filter(Boolean).join('');
}
const nowhite = s => String(s).replace(/\s+/g, '');

let dupCount = 0;
for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  for (const q of (Array.isArray(json) ? json : json.questions || [])) {
    if (!q.table || !String(q.stem || '').includes('\n')) continue;
    const tail = q.stem.split('\n').slice(1).join('\n');
    const tableNW = nowhite(tableCells(q.table));
    if (tableNW.length >= 8 && nowhite(tail).includes(tableNW)) {
      dupCount++;
      console.log(`${file} Q${q.qnum}: stem에 table과 중복되는 원자료 텍스트 있음`);
    }
  }
}
console.log(`\n합계 ${dupCount}건`);
NODE
```

적발된 문항은 `stem`에서 표 내용과 겹치는 부분만 제거한다. `(단, ...)` 단서 문장·SQL 구문처럼 표와 무관한 내용은 보존한다. 표가 두 부분(전/후 비교 등)으로 나뉘는데 `table` 필드가 한쪽만 담고 있다면, 무작정 지우지 말고 **표 필드 자체를 확장**해 원자료 손실을 막는다.

## 제4조의4 자료 필요 문항 image/table 누락 후보 점검

> 제4조의 이미지 파일 누락 확인은 `image` 필드가 이미 있는 문항의 실제 PNG 존재 여부만 본다. 따라서 `stem`이 "다음 트리/그림/표"처럼 PDF 자료를 전제로 하는데 `image`·`table` 필드 자체가 없는 문항은 잡지 못한다. 실제로 `exam_2016-1.json` Q5는 원본 PDF에 트리 그림이 있었지만, PDF 인쇄번호가 `10.`으로 잡히는 보정 매핑 구간이라 자동 추출기가 JSON Q5에 연결하지 못했고, `image` 필드가 없어서 기존 제4조를 통과했다(판례_실패_PDF인쇄번호_JSON큐넘매핑이미지누락_012).
>
> 이 조항은 **즉시 실패 판정이 아니라 PDF 육안 대조 후보 목록화**다. 그래프의 방향 간선이나 표 데이터가 `stem` 안에 완전히 전사된 문항은 정상일 수 있으므로, 후보는 반드시 PDF 원본과 대조한 뒤 누락 여부를 확정한다.

```bash
node - <<'NODE'
const fs = require('fs');
const path = require('path');
const skip = new Set(['categories_index.json', 'exams_index.json', 'cbt_store.json']);

const mediaCue = /(다음\s*(그림|도표|표|트리|그래프|다이어그램|회로|네트워크|구조도|상태도|블록도)(?:에서|의|를|을|에|는|와|과|\s)|그림과\s*같은|표와\s*같은)/i;
const textCompleteHint = /(방향\s*간선\s*:|행렬\s*표기\s*기준|노드\s*정보\s*:|간선\s*정보\s*:|표기\s*기준\s*:)/;
const suspects = [];

for (const file of fs.readdirSync('data').filter(f => f.endsWith('.json') && !skip.has(f)).sort()) {
  const json = JSON.parse(fs.readFileSync(path.join('data', file), 'utf8'));
  const questions = Array.isArray(json) ? json : json.questions || [];
  for (const q of questions) {
    const stem = String(q.stem || '').replace(/\s+/g, ' ').trim();
    if (mediaCue.test(stem) && !textCompleteHint.test(stem) && !q.image && !q.table) {
      suspects.push(`${file} Q${q.qnum}: ${stem.slice(0, 120)}`);
    }
  }
}

if (suspects.length) {
  console.log('자료 필요 문항 image/table 누락 후보:');
  console.log(suspects.join('\n'));
  console.log(`\n합계 ${suspects.length}건`);
  process.exit(1);
}
console.log('자료 필요 문항 image/table 누락 후보 없음');
NODE
```

후보가 나오면 다음 순서로 확정한다.

1. 고시_추출삽입_PDF매핑_저장규칙 제1조의2에 따라 PDF 인쇄번호가 아니라 `stem` 핵심 문구로 JSON `qnum`을 확정한다.
2. PDF 자료가 실제로 있고 `stem`·`options`만으로 풀 수 없으면 `image` 또는 `table`을 추가한다.
3. `stem`에 자료 내용이 완전히 전사되어 있어 이미지가 필요 없는 경우에는 보고서에 정상 예외로 기록한다.
4. 수정 후 제4조(경로·HTML), 제4조의4(누락 후보), API 응답, 앱 렌더링을 재확인한다.
